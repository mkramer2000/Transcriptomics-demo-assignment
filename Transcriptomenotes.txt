#unzip fastq files 
gunzip fastq/*.gz

#build index 
hisat2-build reference.fna index_name

#create sam file 
hisat2 -x index_name -1 fastq/sample_F.fastq -2 fastq/sample_R.fastq -S sample.sam

#convert sam to bam 
samtools view -bS sample.sam > sample.bam

#sort bam file 
samtools sort sample.bam -o sample.sorted.bam

#assemble transcripts 
stringtie sample.sorted.bam -G annotation.gff -o stringtie/sample.transcripts.gtf

#create directory for ballgown output 
mkdir -p ballgown/sample

#generate expression files for ballgown 
stringtie -e -B -p 8 -G stringtie_merged.gtf -o ballgown/sample/sample.gtf sample.sorted.bam
